class Solution:
    def minCostClimbingStairs(self, cost: List[int]) -> int:
        n = len(cost)
        dp = [0] * n
        for i in range(2,n):
            two = dp[i-2] + cost[i-2]
            one = dp[i-1] + cost[i-1]
            dp[i] = min(two,one)
        two = dp[-2] + cost[-2]
        one = dp[-1] + cost[-1]
        final = min(two,one)
        return final
